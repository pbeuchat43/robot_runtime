cmake_minimum_required(VERSION 3.20)
project(robot_runtime_recording VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(rgbd_camera_interfaces REQUIRED)
find_package(rgbd_object_detector_interfaces 0.2.0 REQUIRED)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GSTREAMER REQUIRED IMPORTED_TARGET gstreamer-app-1.0>=1.20)
find_path(NLOHMANN_JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
add_library(recording_core src/recording.cpp)
target_include_directories(recording_core PUBLIC include)
target_include_directories(recording_core SYSTEM PUBLIC ${NLOHMANN_JSON_INCLUDE_DIR})
target_link_libraries(recording_core PUBLIC PkgConfig::GSTREAMER Threads::Threads)
ament_target_dependencies(recording_core PUBLIC sensor_msgs)
add_executable(recording_node src/node.cpp)
target_link_libraries(recording_node PRIVATE recording_core)
ament_target_dependencies(recording_node PUBLIC rclcpp rgbd_camera_interfaces rgbd_object_detector_interfaces)
install(TARGETS recording_core recording_node ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_recording test/test_recording.cpp)
  target_link_libraries(test_recording PRIVATE recording_core)
  add_test(NAME recording_contract COMMAND test_recording)
  add_executable(test_retention test/test_retention.cpp)
  target_link_libraries(test_retention PRIVATE recording_core)
  add_test(NAME recording_retention COMMAND test_retention)
  set_tests_properties(recording_retention PROPERTIES TIMEOUT 10)
endif()
ament_package()
